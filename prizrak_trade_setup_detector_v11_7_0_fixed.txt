f_fmt(float p) => str.tostring(p, format.mintick)
f_clamp_int(int v, int lo, int hi) => math.max(lo, math.min(hi, v))

f_box(box _bx, int _x1, float _y1, int _x2, float _y2, color _bg, color _bd, int _bw) =>
    box out = _bx
    if na(out)
        out := box.new(_x1, _y1, _x2, _y2, bgcolor=_bg, border_color=_bd, border_width=_bw)
    else
        box.set_left(out, _x1)
        box.set_right(out, _x2)
        box.set_top(out, _y1)
        box.set_bottom(out, _y2)
        box.set_bgcolor(out, _bg)
        box.set_border_color(out, _bd)
        box.set_border_width(out, _bw)
    out

f_line(line _ln, int _x1, float _y1, int _x2, float _y2, color _c, int _w) =>
    line out = _ln
    if na(out)
        out := line.new(_x1, _y1, _x2, _y2, color=_c, width=_w)
    else
        line.set_xy1(out, _x1, _y1)
        line.set_xy2(out, _x2, _y2)
        line.set_color(out, _c)
        line.set_width(out, _w)
    out

f_label(label _lb, int _x, float _y, string _txt, label_style _style, color _bg, color _tc, size _sz) =>
    label out = _lb
    if na(out)
        out := label.new(_x, _y, _txt, style=_style, color=_bg, textcolor=_tc, size=_sz)
    else
        label.set_xy(out, _x, _y)
        label.set_text(out, _txt)
        label.set_style(out, _style)
        label.set_color(out, _bg)
        label.set_textcolor(out, _tc)
        label.set_size(out, _sz)
    out

f_bar_price(int off) =>
    float p = na
    if poc_price_mode == "CLOSE"
        p := close[off]
    else if poc_price_mode == "HL2"
        p := (high[off] + low[off]) / 2.0
    else
        p := (high[off] + low[off] + close[off]) / 3.0
    p

f_bin_size(float lo, float hi) =>
    float rng = hi - lo
    float min_bin = math.max(rng * bin_min_pct, syminfo.mintick)
    math.max(rng / poc_bins, min_bin)

calculatePOC(int start_bar, int end_bar, float lo, float hi) =>
    float rng = hi - lo
    if rng <= 0 or poc_bins <= 1
        na
    else
        float bsz = f_bin_size(lo, hi)
        int bins_actual = int(math.max(2, math.ceil((hi - lo) / bsz)))
        array<float> vols = array.new_float(bins_actual, 0.0)
        float cum_pv = 0.0
        float cum_v = 0.0
        for b = start_bar to end_bar
            int off = bar_index - b
            if off >= 0
                float bv = volume[off]
                if not na(bv) and bv > 0
                    float bp = na
                    if poc_price_mode == "VWAP"
                        float tp = (high[off] + low[off] + close[off]) / 3.0
                        cum_pv += tp * bv
                        cum_v += bv
                        bp := cum_v > 0 ? (cum_pv / cum_v) : tp
                    else
                        bp := f_bar_price(off)
                    if not na(bp) and bp >= lo and bp <= hi
                        int idx = int(math.floor((bp - lo) / bsz))
                        idx := f_clamp_int(idx, 0, bins_actual - 1)
                        array.set(vols, idx, array.get(vols, idx) + bv)
        float maxv = 0.0
        float poc = na
        for i = 0 to bins_actual - 1
            float v = array.get(vols, i)
            if v > maxv
                maxv := v
                poc := lo + (i + 0.5) * bsz
        poc

// =========================
// ACCUMULATION

cand_high = ta.highest(high, max_flat_bars)
cand_low = ta.lowest(low, max_flat_bars)
cand_rng = cand_high - cand_low
cand_rng_pct = cand_rng / math.max(cand_low, syminfo.mintick)

cand_touch_top = 0
cand_touch_bot = 0
if enable_accum and not in_flat and bar_index >= max_flat_bars
    for i = 0 to max_flat_bars - 1
        cand_touch_top += high[i] >= cand_high * (1 - touch_tolerance) ? 1 : 0
        cand_touch_bot += low[i] <= cand_low * (1 + touch_tolerance) ? 1 : 0

cand_ok = enable_accum and not in_flat and bar_index >= max_flat_bars and cand_rng_pct <= flat_max_range_pct and cand_touch_top + cand_touch_bot >= min_touches and cand_touch_top >= min_touches_side and cand_touch_bot >= min_touches_side
if cand_ok
    in_flat := true
    flat_start := bar_index - max_flat_bars + 1
    flat_high := cand_high
    flat_low := cand_low
    touches_top := cand_touch_top
    touches_bottom := cand_touch_bot
    exit_confirm_cnt := 0

if in_flat
    outside_early = high > flat_high * (1 + flat_break_early) or low < flat_low * (1 - flat_break_early)
    outside_true = high > flat_high * (1 + flat_break_true) or low < flat_low * (1 - flat_break_true)
    outside_any = close > flat_high or close < flat_low

    touch_top = high >= flat_high * (1 - touch_tolerance) and high <= flat_high * (1 + touch_tolerance)
    touch_bot = low >= flat_low * (1 - touch_tolerance) and low <= flat_low * (1 + touch_tolerance)
    touches_top += touch_top ? 1 : 0
    touches_bottom += touch_bot ? 1 : 0

    if outside_any and not outside_early
        exit_confirm_cnt += 1
    else
        exit_confirm_cnt := 0

    bool finalize = outside_early or outside_true or exit_confirm_cnt >= exit_confirm_bars or (bar_index - flat_start) > max_flat_bars
    if finalize
        bool flat_valid = (touches_top + touches_bottom) >= min_touches
        if enable_poc and flat_valid
            float fpoc = calculatePOC(flat_start, math.max(flat_start, bar_index - 1), flat_low, flat_high)
            if not na(fpoc)
                array.push(poc_prices, fpoc)
                array.push(poc_bars, bar_index)
                array.push(poc_tests, 0)
                if array.size(poc_prices) > max_poc_keep
                    array.shift(poc_prices)
                    array.shift(poc_bars)
                    array.shift(poc_tests)
        last_flat_high := flat_high
        last_flat_low := flat_low
        last_flat_valid := flat_valid
        in_flat := false
        flat_start := na
        flat_high := na
        flat_low := na
        touches_top := 0
        touches_bottom := 0
        exit_confirm_cnt := 0

// =========================
// STOP VOLUME
// =========================
s_hi = ta.highest(high, stop_len)
s_lo = ta.lowest(low, stop_len)
s_rng = s_hi - s_lo
s_rng_pct = s_rng / math.max(s_lo, syminfo.mintick)
s_vol_sum = math.sum(volume, stop_len)

if enable_stop_vol and not in_stop and bar_index >= stop_len
    bool rng_ok = s_rng_pct <= stop_max_range_pct and s_rng <= atr * stop_range_atr_mult
    bool vol_ok = s_vol_sum >= vol_ma * stop_vol_mult * stop_len
    bool flat_ok = not in_flat or stop_detect_inside
    if rng_ok and vol_ok and flat_ok
        in_stop := true
        stop_start := bar_index - stop_len + 1
        stop_high := s_hi
        stop_low := s_lo
        stop_exit_cnt := 0

if in_stop
    stop_out_early = high > stop_high * (1 + stop_break_early) or low < stop_low * (1 - stop_break_early)
    stop_out_any = close > stop_high or close < stop_low
    if stop_out_any and not stop_out_early
        stop_exit_cnt += 1
    else
        stop_exit_cnt := 0
    if stop_out_early or stop_exit_cnt >= stop_exit_confirm
        if enable_poc
            sv_poc = calculatePOC(stop_start, math.max(stop_start, bar_index - 1), stop_low, stop_high)
            if not na(sv_poc)
                array.push(poc_prices, sv_poc)
                array.push(poc_bars, bar_index)
                array.push(poc_tests, 0)
        in_stop := false
        stop_start := na
        stop_high := na
        stop_low := na
        stop_exit_cnt := 0

// CHOCH
// =========================
[htf_ph, htf_pl] = request.security(syminfo.tickerid, htf, [ta.pivothigh(high, choch_pivot_len, choch_pivot_len), ta.pivotlow(low, choch_pivot_len, choch_pivot_len)], lookahead=barmerge.lookahead_off)
var array<float> htf_highs = array.new_float()
var array<float> htf_lows = array.new_float()
if not na(htf_ph)
    array.push(htf_highs, htf_ph)
    if array.size(htf_highs) > 10
        array.shift(htf_highs)
if not na(htf_pl)
    array.push(htf_lows, htf_pl)
    if array.size(htf_lows) > 10
        array.shift(htf_lows)

htf_ready = array.size(htf_highs) >= 2 and array.size(htf_lows) >= 2
htf_trend_up = false
htf_trend_down = false
htf_last_hh = na
htf_last_ll = na
if htf_ready
    htf_last_hh := array.get(htf_highs, array.size(htf_highs) - 1)
    htf_prev_hh = array.get(htf_highs, array.size(htf_highs) - 2)
    htf_last_ll := array.get(htf_lows, array.size(htf_lows) - 1)
    htf_prev_ll = array.get(htf_lows, array.size(htf_lows) - 2)
    htf_trend_up := htf_last_ll > htf_prev_ll
    htf_trend_down := htf_last_hh < htf_prev_hh

if enable_choch and htf_ready
    bool choch_cross_dn = ta.crossunder(close, htf_last_ll * (1 - choch_break_early))
    bool choch_cross_up = ta.crossover(close, htf_last_hh * (1 + choch_break_early))
    choch_active := (htf_trend_up and choch_cross_dn) or (htf_trend_down and choch_cross_up)
    choch_kind := htf_trend_up and choch_cross_dn ? 2 : htf_trend_down and choch_cross_up ? 4 : 0

// =========================
// WORKING LEVEL + TESTS + TRAPS
// =========================
if enable_poc and array.size(poc_prices) > 0
    for i = 0 to array.size(poc_prices) - 1
        lvl = array.get(poc_prices, i)
        tests = array.get(poc_tests, i)
        if bar_index > array.get(poc_bars, i)
            hiTol = lvl * (1 + poc_test_tolerance)
            loTol = lvl * (1 - poc_test_tolerance)
            touched = high >= loTol and low <= hiTol
            if touched
                array.set(poc_tests, i, tests + 1)

float working_level = na
if array.size(poc_prices) > 0
    for j = 0 to array.size(poc_prices) - 1
        i = array.size(poc_prices) - 1 - j
        if array.get(poc_tests, i) < poc_dead_tests
            working_level := array.get(poc_prices, i)
            break

bool work_cross_up = not na(working_level) ? ta.crossover(close, working_level) : false
bool work_cross_dn = not na(working_level) ? ta.crossunder(close, working_level) : false

if enable_traps and not na(working_level)
    brk_up = close > working_level and close[1] <= working_level and volume > vol_ma * 1.2
    brk_dn = close < working_level and close[1] >= working_level and volume > vol_ma * 1.2
    if brk_up
        last_break_bar := bar_index
        last_break_dir := "up"
    if brk_dn
        last_break_bar := bar_index
        last_break_dir := "down"

trap_up = false
trap_down = false
if not na(last_break_bar) and not na(working_level)
    in_window = (bar_index - last_break_bar) <= trap_max_bars
    trap_up := last_break_dir == "up" and in_window and close < working_level and volume < vol_ma * trap_vol_drop
    trap_down := last_break_dir == "down" and in_window and close > working_level and volume < vol_ma * trap_vol_drop

// =========================
// FILTERS + SETUPS
// =========================
hidden_bull  = rsi > rsi[1] and low < low[1]
hidden_bear  = rsi < rsi[1] and high > high[1]
classic_bull = rsi < rsi[1] and low < low[1]
classic_bear = rsi > rsi[1] and high > high[1]
macd_bull = macd_line > macd_signal
macd_bear = macd_line < macd_signal

bull_ok = not enable_filters or ((hidden_bull or classic_bull or macd_bull) and not hidden_bear and not classic_bear and not macd_bear)
bear_ok = not enable_filters or ((hidden_bear or classic_bear or macd_bear) and not hidden_bull and not classic_bull and not macd_bull)

float atr_ref = atr
near_level = not na(working_level) ? math.abs(close - working_level) <= atr_ref * test_atr_frac : false
buy_setup = not choch_active and work_cross_up and (htf_trend_up or allow_counter) and bull_ok and near_level
sell_setup = not choch_active and work_cross_dn and (htf_trend_down or allow_counter) and bear_ok and near_level

open_buy = buy_setup
open_sell = sell_setup
if rr_gate_enabled and (open_buy or open_sell)
    st_buy = not na(last_flat_low) and last_flat_valid ? last_flat_low : close - atr
    st_sell = not na(last_flat_high) and last_flat_valid ? last_flat_high : close + atr
    risk_buy = math.abs(close - st_buy)
    risk_sell = math.abs(close - st_sell)
    rr_buy = risk_buy > 0 ? (risk_buy * rr_main_mult) / risk_buy : na
    rr_sell = risk_sell > 0 ? (risk_sell * rr_main_mult) / risk_sell : na
    open_buy := open_buy and (not na(rr_buy) ? rr_buy >= rr_min : false)
    open_sell := open_sell and (not na(rr_sell) ? rr_sell >= rr_min : false)

if not position_active and (open_buy or open_sell)
    position_active := true
    position_dir := open_buy ? "BUY" : "SELL"
    entry_price := close
    stop_price := position_dir == "BUY" ? (not na(last_flat_low) and last_flat_valid ? last_flat_low : close - atr) : (not na(last_flat_high) and last_flat_valid ? last_flat_high : close + atr)
    risk = math.abs(entry_price - stop_price)
    tp_main := position_dir == "BUY" ? entry_price + risk * rr_main_mult : entry_price - risk * rr_main_mult
    tp_ext := position_dir == "BUY" ? entry_price + risk * rr_ext_mult : entry_price - risk * rr_ext_mult
    breakeven_price := na

if position_active
    risk = math.abs(entry_price - stop_price)
    tp1 = position_dir == "BUY" ? entry_price + risk : entry_price - risk
    tp1_hit = position_dir == "BUY" ? high >= tp1 : low <= tp1
    if tp1_hit and na(breakeven_price)
        breakeven_price := entry_price
        stop_price := entry_price

    tp_main_hit = position_dir == "BUY" ? high >= tp_main : low <= tp_main
    tp_ext_hit = position_dir == "BUY" ? high >= tp_ext : low <= tp_ext
    stop_hit = position_dir == "BUY" ? low <= stop_price : high >= stop_price

    if tp_ext_hit or tp_main_hit or stop_hit
        position_active := false
        position_dir := ""
        entry_price := na
        stop_price := na
        tp_main := na
        tp_ext := na
        breakeven_price := na
